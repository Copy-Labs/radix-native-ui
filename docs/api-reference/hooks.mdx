# Hooks

Radix Themes Native provides several hooks for accessing theme values and controlling theme behavior.

## useTheme

Returns the current theme object with all design tokens.

### Usage

```tsx
import { useTheme } from 'radix-native-ui';

function MyComponent() {
  const theme = useTheme();

  return (
    <View style={{
      backgroundColor: theme.colors.gray[2],
      padding: theme.space[4]
    }}>
      <Text style={{ color: theme.colors.gray[12] }}>
        Themed content
      </Text>
    </View>
  );
}
```

### Returns

```tsx
interface Theme {
  name: string;
  scaling: number;
  radius: 'none' | 'small' | 'medium' | 'large' | 'full';
  grayColor: 'mauve' | 'olive' | 'sage' | 'sand' | 'slate' | 'gray';
  accentColor: Color;
  colors: ColorTokens; // gray, blue, ..., blackAlpha, whiteAlpha
  space: SpaceScale;
  radii: RadiusScale;
  typography: TypographyTokens;
  shadows: { light: ShadowTokens; dark: ShadowTokens };
}
```

Typography values — including `fontSizes`, `fontWeights`, and `fonts` — live under `theme.typography`. There are no top-level `fontSizes`, `fontWeights`, or `fonts` fields on `Theme`.

## useThemeMode

Returns the current theme mode ('light' or 'dark').

### Usage

```tsx
import { useThemeMode } from 'radix-native-ui';

function MyComponent() {
  const mode = useThemeMode();

  return (
    <Text>Current mode: {mode}</Text>
  );
}
```

### Returns

```tsx
type ThemeMode = 'light' | 'dark';
```

---

## useThemeActions

Returns theme control functions and the current mode.

### Usage

```tsx
import { useThemeActions } from 'radix-native-ui';

function ThemeToggle() {
  const { setMode, toggleMode, mode } = useThemeActions();

  return (
    <Flex direction="column" gap={8}>
      <Text>Current: {mode}</Text>
      <Button onPress={toggleMode}>Toggle Theme</Button>
      <Button onPress={() => setMode('dark')}>Dark Mode</Button>
      <Button onPress={() => setMode('light')}>Light Mode</Button>
    </Flex>
  );
}
```

### Returns

```tsx
interface ThemeActions {
  mode: ThemeMode;
  setMode: (mode: ThemeMode) => void;
  toggleMode: () => void;
}
```

## useThemeColors

Returns convenience colors for the current mode.

### Usage

```tsx
import { useThemeColors } from 'radix-native-ui';

function MyComponent() {
  const { gray, background, surface, panel } = useThemeColors();

  return (
    <View style={{ backgroundColor: background }}>
      <View style={{ backgroundColor: surface }}>
        <Text style={{ color: gray[12] }}>Content</Text>
      </View>
    </View>
  );
}
```

### Returns

```tsx
interface ThemeColors {
  gray: ColorScale;
  background: string;
  surface: string;
  panel: string;
}
```

## useHaptics

Returns the global haptics setting from `ThemeProvider` as a boolean (`true` by default). Throws if called outside a `ThemeProvider`.

### Usage

```tsx
import { useHaptics, Text } from 'radix-native-ui';

function HapticsSettingLabel() {
  const hapticsEnabled = useHaptics();

  return <Text>{hapticsEnabled ? 'Haptics on' : 'Haptics off'}</Text>;
}
```

### Returns

```tsx
type Haptics = boolean;
```

<Note>
  A haptic fires only when the global `haptics` setting is `true` **and** the individual component's `hapticFeedback` prop is enabled. See [ThemeProvider](/api-reference/theme-provider) for the `haptics` prop.
</Note>

## useToast

Returns the toast notification controller.

### Usage

```tsx
import { useToast } from 'radix-native-ui';

function MyComponent() {
  const { show, success } = useToast();

  const handleSuccess = () => {
    success({
      title: 'Success!',
      description: 'Your changes have been saved.',
    });
    // Or use the generic form: show({ title: 'Saved', variant: 'success' })
  };

  return (
    <Button onPress={handleSuccess}>Save Changes</Button>
  );
}
```

### Returns

```tsx
interface ToastController {
  show: (options: ToastOptions) => string;
  success: (options: Omit<ToastOptions, 'variant'>) => string;
  error: (options: Omit<ToastOptions, 'variant'>) => string;
  warning: (options: Omit<ToastOptions, 'variant'>) => string;
  info: (options: Omit<ToastOptions, 'variant'>) => string;
  accent: (options: Omit<ToastOptions, 'variant'>) => string;
  hide: (id: string) => void;
  hideAll: () => void;
  update: (id: string, options: Partial<ToastOptions>) => void;
}

interface ToastOptions {
  title: string;
  description?: string;
  variant?: 'default' | 'success' | 'error' | 'warning' | 'info' | 'accent';
  duration?: number;
  action?: {
    label: string;
    onPress: () => void;
  };
  // Also accepts color, position, icon, showClose, style, and callbacks.
}
```

The `show`, `success`, `error`, `warning`, `info`, and `accent` methods return the toast ID, which you can pass to `hide` or `update`. Use the `variant` field (not `type`) to set a toast's visual style.

## Color Helper Functions

### getVariantColors

Get colors for a specific variant with highContrast support.

```tsx
import { getVariantColors, useTheme, useThemeMode } from 'radix-native-ui';

function CustomButton({ variant = 'solid', highContrast = false }) {
  const theme = useTheme();
  const mode = useThemeMode();

  const colors = getVariantColors(
    theme,
    'blue',      // color name
    mode,
    variant,     // 'solid' | 'soft' | 'outline' | 'ghost' | 'classic'
    highContrast
  );

  // Returns: { backgroundColor, textColor, borderColor }
}
```

### getHighContrastTextColor

Returns the high contrast text color.

```tsx
import { getHighContrastTextColor, useTheme } from 'radix-native-ui';

const theme = useTheme();
const textColor = getHighContrastTextColor(theme);
// Returns: blackAlpha[12]
```

### getHighContrastBackground

Returns the high contrast background color.

```tsx
import { getHighContrastBackground, useTheme, useThemeMode } from 'radix-native-ui';

const theme = useTheme();
const mode = useThemeMode();
const bgColor = getHighContrastBackground(theme, 'blue', mode);
// Returns: blue[11]
```

### getColorScale

Returns the color scale for any color name.

```tsx
import { getColorScale, useTheme, useThemeMode } from 'radix-native-ui';

const theme = useTheme();
const mode = useThemeMode();
const scale = getColorScale(theme, 'blue', mode);
// Returns: ColorScale object with keys 1-12
```

### getColorAlpha

Returns the alpha color scale for any color name.

```tsx
import { getColorAlpha, useTheme } from 'radix-native-ui';

const theme = useTheme();
const alpha = getColorAlpha(theme, 'blue');
// Returns: AlphaColorScale object with keys 1-12
```
